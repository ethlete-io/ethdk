import { Component, computed, input, StaticProvider } from '@angular/core';
import { createOverlayOpener, OverlayDefinition, TableColumnVisibilityHost } from '@ethlete/components';

@Component({ selector: 'app-a-edit-dialog', template: '' })
class EditDialogComponent {}

declare const EDIT_A_DIALOG: OverlayDefinition<EditDialogComponent, boolean>;
declare const EDIT_B_DIALOG: OverlayDefinition<EditDialogComponent, boolean>;

@Component({ selector: 'app-a-detail', template: '' })
export class DetailComponent {
  public table = input<TableColumnVisibilityHost | undefined>(undefined);

  protected hasHiddenColumns = computed(() => {
    const table = this.table();

    return !!table && table.visibleColumnsMeta().length < table.allColumns().length;
  });

  protected columnCount = computed(() => this.table()?.allColumns().length ?? 0);

  private refreshOnSave = {
    afterClosed: (didSave: boolean | null) => {
      if (didSave) this.refresh();
    },
  };

  private editAOpener = createOverlayOpener(EDIT_A_DIALOG, this.refreshOnSave);
  private editBOpener = createOverlayOpener(EDIT_B_DIALOG, this.refreshOnSave);

  protected openEditA(providers: StaticProvider[] | null) {
    if (providers) this.editAOpener.open({ providers });
  }

  protected openEditB(providers: StaticProvider[]) {
    this.editBOpener.open({ providers });
  }

  private refresh() {
    console.log('refresh');
  }
}
