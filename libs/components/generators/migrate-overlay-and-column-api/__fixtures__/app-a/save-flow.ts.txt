import { Component } from '@angular/core';
import {
  createOverlayOpener,
  injectCommandPalette,
  OverlayDefinition,
  TableColumnVisibilityHost,
  TableFeatureHost,
} from '@ethlete/components';

@Component({ selector: 'app-a-save-dialog', template: '' })
class SaveDialogComponent {}

@Component({ selector: 'app-a-pick-dialog', template: '' })
class PickDialogComponent {}

declare const SAVE_DIALOG: OverlayDefinition<SaveDialogComponent, boolean>;
declare const PICK_DIALOG: OverlayDefinition<PickDialogComponent, string>;
declare const table: TableFeatureHost;

@Component({ selector: 'app-a-save-flow', template: '' })
export class SaveFlowComponent {
  protected palette = injectCommandPalette();

  protected saveDialog = createOverlayOpener(SAVE_DIALOG, {
    afterClosed: (didSave: boolean | null) => {
      if (didSave === null) return;
      this.saved = didSave;
    },
    beforeClosed: (didSave: boolean | null) => this.log(didSave !== null),
  });

  protected pickDialog = createOverlayOpener(PICK_DIALOG, { afterClosed: this.onPicked });

  protected columnHost: TableColumnVisibilityHost = {
    allColumns: () => table.allColumns(),
    visibleColumnsMeta: () => table.visibleColumnsMeta(),
    isColumnVisible: (key) => table.isColumnVisible(key),
    setColumnVisible: (key, visible) => table.setColumnVisible(key, visible),
    showAllColumns: () => table.showAllColumns(),
    resolvedLabels: () => table.resolvedLabels(),
  };

  protected saved = false;
  protected picked: string | null = null;

  protected openPick() {
    this.pickDialog.open({
      afterClosed(choice: string | null) {
        if (choice !== null) console.log(choice);
      },
    });
  }

  private onPicked(choice: string | null) {
    this.picked = choice ?? null;
  }

  private log(value: boolean) {
    console.log(value);
  }
}
