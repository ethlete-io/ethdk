import { definePart, drawing, html, svgAsset } from '@design-system';
import mark from '@assets/logo.svg?raw';
import styles from './logo.css?inline';

export const tomflixMark = (size: number) => svgAsset(mark, { size, attrs: { 'aria-hidden': 'true' } });

type View = 'mark' | 'anatomy' | 'clear-space' | 'minimum-size' | 'colour-variants' | 'misuse' | 'motion';

const TITLES: Record<View, string> = {
  mark: '',
  anatomy: 'Aufbau',
  'clear-space': 'Schutzraum',
  'minimum-size': 'Mindestgröße',
  'colour-variants': 'Farbvarianten',
  misuse: 'Falsche Verwendung',
  motion: 'Bewegung',
};

export default definePart<{ size: number; view: View }>({
  note: 'Das Tomflix-Zeichen: Das gefaltete T aus dem alten Katalog, nur geschärft.',
  styles,
  states: {
    mark: { size: 160, view: 'mark' },
    anatomy: { size: 96, view: 'anatomy' },
    'clear-space': { size: 96, view: 'clear-space' },
    'minimum-size': { size: 16, view: 'minimum-size' },
    'colour-variants': { size: 96, view: 'colour-variants' },
    misuse: { size: 96, view: 'misuse' },
    motion: { size: 96, view: 'motion' },
  },
  describe: {
    mark: 'The mark alone at one size on the dark surface; judge its shape and whether it reads as a T.',
    anatomy: 'Placeholder until the pick: the mark with the section title; the parts are named after the merge.',
    'clear-space':
      'Placeholder until the pick: the mark with the section title; the clear space is drawn after the merge.',
    'minimum-size': 'The mark at 16 px; check that it still reads as a T.',
    'colour-variants': 'Placeholder until the pick: the mark with the section title; colour variants follow the merge.',
    misuse: 'Placeholder until the pick: the mark with the section title; misuse examples follow the merge.',
    motion: 'Placeholder until the pick: the mark with the section title; the motion follows the merge.',
  },
  render: ({ size, view }) =>
    drawing({
      body:
        view === 'mark'
          ? html`<div class="logo-stage et-surface--dark">${tomflixMark(size)}</div>`
          : html`<section class="logo-stage et-surface--dark">
              <h2 class="logo-title">${TITLES[view]}</h2>
              ${tomflixMark(size)}
            </section>`,
    }),
});
