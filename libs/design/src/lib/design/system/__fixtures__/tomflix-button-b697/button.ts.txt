import { definePart, drawing } from '@design-system';
import { grid, renderButton, states, type ButtonProps } from '@shared/button-states';
import styles from './button.css?inline';

export default definePart<ButtonProps>({
  note: 'Der SDK-Button als Pille mit Tomflix-Rot.',
  states,
  grid,
  controls: { button: { kind: 'button', input: ['pointer', 'keyboard'], disabled: true } },
  describe: {
    default: 'The main action at rest; check the red fill and the label.',
    'default-hover': 'The main action under the pointer; check that the fill lightens.',
    'default-active': 'The main action pressed; check that the fill darkens.',
    'default-focus-visible': 'The main action focused by keyboard; check the focus ring.',
    tonal: 'The second action at rest; check that it sits behind the main one.',
    'tonal-hover': 'The second action under the pointer; check the lighter tone.',
    'tonal-active': 'The second action pressed; check the pressed tone.',
    'tonal-focus-visible': 'The second action focused by keyboard; check the focus ring.',
    outline: 'The third action at rest; check the outline on the dark surface.',
    'outline-hover': 'The third action under the pointer; check the hover fill.',
    'outline-active': 'The third action pressed; check the pressed fill.',
    'outline-focus-visible': 'The third action focused by keyboard; check the focus ring.',
    disabled: 'The main action disabled; check that it reads as not clickable.',
  },
  render: async (props) => drawing({ body: await renderButton(props, 'font-bold'), styles }),
});
